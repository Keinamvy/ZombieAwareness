## Review Focus

- **CoroConfigRegistry file round-tripping with malformed/blank lines or comments**: `CoroConfigRegistry` must ignore comment lines (`#` / `//`), blank lines, and unknown fields when loading `.cfg` files without throwing or resetting valid fields.
- **Unloaded chunks during pathfinding or light/sound checks**: `CoroUtilPath.tryMoveToXYZLongDist` and `ZAUtil.canSpawnTrace` must guard against unloaded positions (`!level.isLoaded(pos)`) and null navigation paths.
- **Mixin target signature mismatches in 26.3**: `ServerLevel.playSeededSound` and `ServerLevel.levelEvent` changed their first parameter from `Player` to `Entity`, `Explosion` split into `ServerExplosion`, and `StartAttacking` uses unobfuscated `lambda$create$3`; every mixin must pass `MixinEnvironment.getCurrentEnvironment().audit()` on the real 26.3 server.
- **EntityScent zero-size / null entity type key in 26.3**: `EntityType.Builder.build(ResourceKey<EntityType<?>>)` requires a `ResourceKey.create(Registries.ENTITY_TYPE, Identifier)` in 26.3 or it throws `NullPointerException` during entity construction.
- **Repeated attribute modifier application on already-boosted zombies**: `ZAUtil.giveRandomSpeedBoost` must check `hasModifier(CoroUtilAttributes.SPEED_BOOST_ID)` and `getBooleanOr(SPEED_BOOST_TAG, false)` so zombies do not crash with duplicate `AttributeModifier` IDs on reload.

package com.corosus.modconfig;

import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.io.TempDir;
import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;
import static org.junit.jupiter.api.Assertions.*;

public class CoroConfigRegistryTest {
    public static class DummyConfig implements IConfigCategory {
        @ConfigComment("Max range")
        public static int maxRange = 64;
        public static boolean enabled = true;
        public static double multiplier = 1.5D;
        public static String whitelist = "PlayerA, PlayerB";
        public boolean hooked = false;
        private final String fileName;
        public DummyConfig(String fileName) { this.fileName = fileName; }
        @Override public String getName() { return "Dummy"; }
        @Override public String getRegistryName() { return "dummy"; }
        @Override public String getConfigFileName() { return fileName; }
        @Override public String getCategory() { return "Dummy"; }
        @Override public void hookUpdatedValues() { hooked = true; }
    }

    @Test
    void writesDefaultAndLoadsModifiedAndIgnoresMalformedLines(@TempDir Path tempDir) throws Exception {
        File cfgFile = tempDir.resolve("Dummy.cfg").toFile();
        DummyConfig cfg = new DummyConfig(cfgFile.getAbsolutePath().replace(".cfg", ""));
        CoroConfigRegistry reg = new CoroConfigRegistry(tempDir.toFile());
        reg.addConfigFile("zombieawareness", cfg);
        assertTrue(cfgFile.exists());
        assertTrue(cfg.hooked);

        Files.writeString(cfgFile.toPath(), "# comment\n\nmalformed_line_without_equals\nunknownField=999\nmaxRange=128\nenabled=false\nmultiplier=2.25\nwhitelist=Bob\n");
        cfg.hooked = false;
        reg.forceLoadRuntimeSettingsFromFile();
        assertEquals(128, DummyConfig.maxRange);
        assertFalse(DummyConfig.enabled);
        assertEquals(2.25D, DummyConfig.multiplier, 0.001D);
        assertEquals("Bob", DummyConfig.whitelist);
        assertTrue(cfg.hooked);
    }
}
